==========Fragment(<></>)==========
- Is a place holder for the parent element.
- Allows us to return 2 DIVS.
***********app.jsx**************
import './App.css'

const Text=()=>{
  return (
    <div>
      <p>Hello World!</p>
    </div>
  );
}

function App() {
  return (
    <>
      <Text/>
      <Text/>
      <Text/>
    </>
  );
}

export default App

*****************app.jsx***********************
import './App.css'

const Text=({display})=>{
  return (
    <div>
      <p>{display}</p>
    </div>
  );
}

function App() {
  return (
    <>
      <Text display="Hello World"/>
      <Text display='Whats Up!'/>
      <Text display='Have a nice day!!'/>
    </>
  );
}

export default App
